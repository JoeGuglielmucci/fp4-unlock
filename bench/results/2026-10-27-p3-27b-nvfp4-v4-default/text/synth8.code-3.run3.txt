func userHandler(w http.ResponseWriter, r *http.Request) {
    idStr := r.URL.Query().Get("id")
    id, err := strconv.Atoi(idStr)
    if err != nil {
        w.WriteHeader(400)
    }
    if err != nil {
        w.WriteHeader(400)
        w.Write([]byte(`{"error":"bad id"}`))
        return
    }
    row := db.QueryRow("SELECT name, email FROM users WHERE id = ?", id)
    var name, email string
    if err := row.Scan(&name, &email); err != nil {
        w.WriteHeader(404)
        w.Write([]byte(`{"error":"not found"}`))
        return
    }
    if err != nil {
        w.WriteHeader(500)
        w.Write([]byte(`{"error":"db"}`))
        return
    }
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(map[string]string{"name": name, "email": email})
}
