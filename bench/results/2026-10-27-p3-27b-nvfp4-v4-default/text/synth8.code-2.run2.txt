def merge_intervals(intervals):
    if not intervals:
        return intervals
    intervals.sort(key=lambda x: x[1])
    merged = [intervals[0]]
    for current in intervals:
        last = merged[-1]
        if current[0] < last[1]:
            last[1] = max(last[1], current[1])
        else:
            merged.append(current)
    return merged
```
